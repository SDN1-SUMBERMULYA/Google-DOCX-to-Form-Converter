    function log(message, type = 'info') {
        const consoleEl = document.getElementById('logConsole');
        const timeStamp = new Date().toLocaleTimeString('id-ID', { hour12: false });
        let colorClass = 'text-slate-300';
        let icon = '<i class="fa-solid fa-chevron-right mr-1.5 text-slate-500"></i>';

        if (type === 'success') {
            colorClass = 'text-emerald-400 font-semibold';
            icon = '<i class="fa-solid fa-circle-check mr-1.5 text-emerald-500"></i>';
        } else if (type === 'error') {
            colorClass = 'text-rose-400 font-bold';
            icon = '<i class="fa-solid fa-circle-xmark mr-1.5 text-rose-500"></i>';
        } else if (type === 'processing') {
            colorClass = 'text-blue-400';
            icon = '<i class="fa-solid fa-arrows-spin animate-spin mr-1.5 text-blue-500"></i>';
        } else if (type === 'custom') {
            colorClass = 'text-emerald-900 !m-0';
            icon = '';
        }

        const logRow = document.createElement('div');
        if (type === 'custom') {
            logRow.className = `${colorClass} `;
            logRow.innerHTML = `<span class="flex font-mono justify-end leading-none text-xs whitespace-pre">${message}</span>`;
        } else if (type === 'processing') {
            logRow.className = `${colorClass} py-0.5 flex items-start`;
            logRow.innerHTML = `<span class="text-slate-600 mr-2 shrink-0">[${timeStamp}]</span> ${icon} <span class="break-all animate-pulse">${message}</span>`;
        } else {
            logRow.className = `${colorClass} py-0.5 flex items-start`;
            logRow.innerHTML = `<span class="text-slate-600 mr-2 shrink-0">[${timeStamp}]</span> ${icon} <span class="break-all">${message}</span>`;
        }

        if (consoleEl.children.length === 1 && consoleEl.children[0].innerText.includes('Menunggu inisiasi')) {
            consoleEl.innerHTML = '';
        }

        consoleEl.appendChild(logRow);
        consoleEl.scrollTop = consoleEl.scrollHeight;

        return logRow;
    }
