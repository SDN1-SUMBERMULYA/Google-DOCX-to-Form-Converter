    function showToast(message, type = 'success') {
        const container = document.getElementById('toastContainer');

        // Buat elemen Toast
        const toast = document.createElement('div');
        toast.className = `flex items-center gap-3 p-3.5 rounded-xl shadow-xl border bg-white dark:bg-[#0c0c0e] text-zinc-800 dark:text-zinc-100 pointer-events-auto transform translate-x-full opacity-0 transition-all duration-300 ease-out border-l-4`;

        let icon = 'info';
        let colorClass = 'border-l-blue-500';
        let iconColor = 'text-blue-500';

        if (type === 'success') {
            icon = 'check-circle';
            colorClass = 'border-l-emerald-500';
            iconColor = 'text-emerald-500';
        } else if (type === 'warning') {
            icon = 'alert-triangle';
            colorClass = 'border-l-amber-500';
            iconColor = 'text-amber-500';
        } else if (type === 'error') {
            icon = 'x-circle';
            colorClass = 'border-l-rose-500';
            iconColor = 'text-rose-500';
        }

        toast.classList.add(colorClass);

        toast.innerHTML = `
                <div class="${iconColor} shrink-0">
                    <i data-lucide="${icon}" class="w-4 h-4"></i>
                </div>
                <div class="flex-1 text-xs font-medium">${message}</div>
                <button class="text-zinc-400 hover:text-zinc-600 dark:hover:text-zinc-200 transition-colors">
                    <i data-lucide="x" class="w-3.5 h-3.5"></i>
                </button>
            `;

        container.appendChild(toast);
        lucide.createIcons();

        // Jalankan animasi slide-in
        setTimeout(() => {
            toast.classList.remove('translate-x-full', 'opacity-0');
        }, 10);

        // Auto remove setelah 4 detik
        const autoRemove = setTimeout(() => {
            dismissToast(toast);
        }, 4000);

        // Event listener tombol close
        toast.querySelector('button').addEventListener('click', () => {
            clearTimeout(autoRemove);
            dismissToast(toast);
        });
    }

    function processSelectedFile(file) {
        if (!file.name.endsWith('.docx')) {
            log(`Peringatan: File yang diunggah harus memiliki format .docx!`, 'error');
            return;
        }

        uploadedFileName = file.name;
        document.getElementById('uploadStatusText').innerHTML = `Berkas Terpilih:<br /><strong class="text-emerald-600 dark:text-slate-300">${file.name}</strong>`;

        const reader = new FileReader();
        reader.onload = function(e) {
            const rawResult = e.target.result;
            uploadedFileBase64 = rawResult.split(',')[1];
            log(`Berhasil membaca berkas ${file.name} secara lokal. Siap diproses ke server cloud.`, 'success');
        };
        reader.readAsDataURL(file);
    }

            google.script.run
                .withSuccessHandler(function(response) {
                    btnSubmit.disabled = false;
                    btnSubmit.className = "w-full px-5 py-3 text-xs font-semibold text-white bg-emerald-600 hover:bg-emerald-500 active:scale-[0.97] border border-emerald-500/30 rounded-xl shadow-md flex items-center justify-center gap-2 transition-all cursor-pointer group";
                    btnSubmit.innerHTML = `<i data-lucide="bot" class="w-4 h-4 group-hover:rotate-12 transition-transform"></i> Generate Form`;
                    lucide.createIcons();

                    if (response.success) {
                        // MODIFIKASI: Ubah log pengiriman payload tadi menjadi SUKSES
                        updateLogStatus(sendingPayloadLog, `Form berhasil dibuat dan diproses oleh Cloud Server!`, 'success');

                        log(`Nama Form Baru Yang Dibuat : "${response.title}"`, 'success');
                        if (response.tempDocTrashed) {
                            log(`Sistem Keamanan : Berkas Google Doc temporer telah dibersihkan dari Google Drive Anda.`, 'success');
                        }
                        log(`Proses Pembuatan Google Form Selesai!`, 'success');
                        showToast(`Proses Pembuatan Google Form Selesai!`, 'success');
                        sdnLogoAscii.forEach(baris => log(baris, 'custom'));

                        document.getElementById('successCard').classList.remove('hidden');
                        document.getElementById('editFormUrl').href = response.editUrl;
                        document.getElementById('lihatFormUrl').href = response.previewUrl;
                        document.getElementById('resFormUrl').href = response.publishedUrl;
                    } else {
                        // MODIFIKASI: Ubah log pengiriman payload menjadi ERROR jika proses gagal
                        updateLogStatus(sendingPayloadLog, `Gagal memproses payload di Cloud Server.`, 'error');

                        log(`Server Error : ${response.message}`, 'error');
                        showToast(`Server Error : ${response.message}`, 'error');
                        document.getElementById('errorCard').classList.remove('hidden');
                        document.getElementById('errorText').innerText = response.message;
                    }
                })
                .withFailureHandler(function(err) {
                    btnSubmit.disabled = false;
                    btnSubmit.className = "w-full px-5 py-3 text-xs font-semibold text-white bg-emerald-600 hover:bg-emerald-500 active:scale-[0.97] border border-emerald-500/30 rounded-xl shadow-md flex items-center justify-center gap-2 transition-all cursor-pointer group";
                    btnSubmit.innerHTML = `<i data-lucide="bot" class="w-4 h-4 group-hover:rotate-12 transition-transform"></i> Generate Form`;
                    lucide.createIcons();

                    // MODIFIKASI: Ubah log pengiriman payload menjadi ERROR jika koneksi gagal
                    updateLogStatus(sendingPayloadLog, `Koneksi gagal saat mengirim payload ke Cloud Server.`, 'error');

                    log(`Koneksi Gagal : ${err.message}`, 'error');
                    showToast(`Proses Pembuatan Google Form Selesai!`, 'success');
                    document.getElementById('errorCard').classList.remove('hidden');
                    document.getElementById('errorText').innerText = err.message;
                })
                .executeFormGeneration(configObj);
                lucide.createIcons();
        } else {
            // Mode Simulasi Browser Lokal
            sendingPayloadLog = log(`[Simulasi] Mengirim payload kustom ke Cloud Server Apps...`, 'processing');

            setTimeout(() => {
                if (configObj.sourceType === 'docxUpload') {
                    log(`[Simulasi] Mengirim berkas ${configObj.docxFileName} ke Drive API converter...`, 'processing');
                    log(`[Simulasi] Berhasil mengonversi berkas DOCX ke format Google Docs (ID: mock-doc-123)`, 'success');
                }
                log(`[Simulasi] Membaca isi dokumen...`, 'processing');
                log(`[Simulasi] Menyusun Form: "${configObj.formTitle || 'Form Kuis Baru'}"`, 'info');
                log(`[Simulasi] Menyinkronkan konfigurasi: Poin PG = ${configObj.pointPG}, Poin Esai= ${configObj.pointIsian}, Email Mode = ${configObj.collectEmail}`, 'info');
                sdnLogoAscii.forEach(baris => log(baris, 'custom'));

                setTimeout(() => {
                    if (configObj.sourceType === 'docxUpload' && configObj.autoCleanGDoc) {
                        log(`[Simulasi] Menghapus file Google Docs temporer secara otomatis (ID: mock-doc-123)`, 'success');
                    }
                    btnSubmit.disabled = false;
                    btnSubmit.className = "w-full px-5 py-3 text-xs font-semibold text-white bg-emerald-600 hover:bg-emerald-500 active:scale-[0.97] border border-emerald-500/30 rounded-xl shadow-md flex items-center justify-center gap-2 transition-all cursor-pointer group";
                    btnSubmit.innerHTML = `<i data-lucide="bot" class="w-4 h-4 group-hover:rotate-12 transition-transform"></i> Generate Form`;
                    lucide.createIcons();

                    // MODIFIKASI SIMULASI: Ubah log pengiriman simulasi menjadi SUKSES
                    updateLogStatus(sendingPayloadLog, `[Simulasi] Payload berhasil dikirim dan diproses secara lokal!`, 'success');

                    document.getElementById('successCard').classList.remove('hidden');
                    document.getElementById('editFormUrl').href = "#";
                    log(`[Simulasi] Berhasil membuat Google Form! Klik tombol di atas untuk melihat hasilnya`, 'success');
                    showToast(`Proses Pembuatan Google Form Selesai!`, 'success');
                }, 1500);
            }, 1000);
        }
    }

    function copyToClipboard(elementId) {
        const link = document.getElementById(elementId).href;
        const tempInput = document.createElement("input");
        tempInput.value = link;
        document.body.appendChild(tempInput);
        tempInput.select();
        document.execCommand('copy');
        log(`Link Google Form disalin ke clipboard!`, 'success');
        log(`${link}`, 'success');
        showToast(`Link Google Form disalin ke clipboard!`, 'success');
        document.body.removeChild(tempInput);
    }
